```python
"""app/services/gemini_chat.py — Chat with provider cascade."""
from __future__ import annotations
import logging, os
from typing import List, Dict, Optional
import httpx

logger = logging.getLogger(__name__)

_GEMINI_BASE = "https://generativelanguage.googleapis.com/v1beta/models"
_GEMINI_CHAT_MODELS = ["gemini-2.0-flash", "gemini-1.5-flash-latest", "gemini-1.5-flash"]

SYSTEM_PROMPT = (
    "You are VIT Assistant, the in-app sports-betting copilot for VIT Sports "
    "Intelligence Network. Help users understand predictions, ML models, ROI/CLV "
    "analytics, the accumulator builder, trust system, validator network, governance, "
    "wallet, training pipeline, AI insights, KYC, subscriptions and developer API. "
    "Reason about football fixtures and interpret model output. Stay grounded in "
    "data given to you. Keep responses concise, conversational and actionable. "
    "Never guarantee bet outcomes or give financial advice."
)

def _no_key_response() -> Dict:
    return {
        "available": False,
        "reply": (
            "No AI provider is configured. Add GEMINI_API_KEY, CLAUDE_API_KEY, "
            "or XAI_API_KEY in Replit Secrets to enable the assistant."
        ),
        "error": "No AI provider configured",
    }

def _build_contents(history: List[Dict], message: str) -> List[Dict]:
    contents = []
    for turn in history[-12:]:
        role = "user" if turn.get("role") == "user" else "model"
        text = (turn.get("content") or "").strip()
        if text:
            contents.append({"role": role, "parts": [{"text": text}]})
    contents.append({"role": "user", "parts": [{"text": message.strip()}]})
    return contents

async def _try_gemini(message: str, history: List[Dict], context: Optional[str]) -> Optional[Dict]:
    api_key = os.getenv("GEMINI_API_KEY", "").strip()
    if not api_key:
        return None
    full_message = f"{context}\n\n{message}" if context else message
    contents = _build_contents(history, full_message)
    system_instruction = {"parts": [{"text": SYSTEM_PROMPT}]}
    for model in _GEMINI_CHAT_MODELS:
        try:
            url = f"{_GEMINI_BASE}/{model}:generateContent?key={api_key}"
            async with httpx.AsyncClient(timeout=20) as client:
                resp = await client.post(url, json={
                    "contents": contents,
                    "systemInstruction": system_instruction,
                    "generationConfig": {"maxOutputTokens": 600, "temperature": 0.7},
                })
            if resp.status_code == 200:
                data = resp.json()
                text = data["candidates"][0]["content"]["parts"][0]["text"]
                return {"available": True, "reply": text.strip(), "error": None, "provider": "gemini"}
            elif resp.status_code == 429:
                logger.warning(f"[gemini] rate limited on {model}, trying next")
                continue
            elif resp.status_code == 400:
                logger.warning(f"[gemini] invalid key")
                return None
        except Exception as exc:
            logger.error(f"[gemini] {model} error: {exc}")
            continue
    return None  # all gemini models failed

async def _try_claude(message: str, history: List[Dict], context: Optional[str]) -> Optional[Dict]:
    api_key = (os.getenv("CLAUDE_API_KEY") or os.getenv("ANTHROPIC_API_KEY") or "").strip()
    if not api_key:
        return None
    try:
        import anthropic
        client = anthropic.AsyncAnthropic(api_key=api_key)
        full_message = f"{context}\n\n{message}" if context else message
        msgs = []
        for turn in history[-8:]:
            role = turn.get("role", "user")
            if role not in ("user", "assistant"):
                role = "user"
            msgs.append({"role": role, "content": turn.get("content", "")})
        msgs.append({"role": "user", "content": full_message})
        response = await client.messages.create(
            model="claude-haiku-4-20250514",
            max_tokens=600,
            system=SYSTEM_PROMPT,
            messages=msgs,
        )
        return {"available": True, "reply": response.content[0].text, "error": None, "provider": "claude"}
    except Exception as exc:
        logger.error(f"[claude] chat error: {exc}")
        return None

async def _try_grok(message: str, history: List[Dict], context: Optional[str]) -> Optional[Dict]:
    api_key = os.getenv("XAI_API_KEY", "").strip()
    if not api_key:
        return None
    try:
        full_message = f"{context}\n\n{message}" if context else message
        msgs = [{"role": "system", "content": SYSTEM_PROMPT}]
        for turn in history[-8:]:
            msgs.append({"role": turn.get("role","user"), "content": turn.get("content","")})
        msgs.append({"role": "user", "content": full_message})
        async with httpx.AsyncClient(timeout=20) as client:
            resp = await client.post(
                "https://api.x.ai/v1/chat/completions",
                headers={"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"},
                json={"model": "grok-beta", "messages": msgs, "max_tokens": 600},
            )
        if resp.status_code == 200:
            text = resp.json()["choices"][0]["message"]["content"]
            return {"available": True, "reply": text.strip(), "error": None, "provider": "grok"}
    except Exception as exc:
        logger.error(f"[grok] chat error: {exc}")
    return None

async def _try_puter(message: str, history: List[Dict], context: Optional[str]) -> Optional[Dict]:
    """Puter free Claude access — no API key needed server-side."""
    try:
        full_message = f"{SYSTEM_PROMPT}\n\nContext: {context or 'General VIT question'}\n\nUser: {message}"
        async with httpx.AsyncClient(timeout=25) as client:
            resp = await client.post(
                "https://api.puter.com/drivers/call",
                json={
                    "interface": "puter-chat-completion",
                    "driver": "claude-claude-3-5-haiku",
                    "test_mode": False,
                    "call": {
                        "method": "complete",
                        "args": {
                            "messages": [{"role": "user", "content": full_message}],
                            "max_tokens": 600,
                        },
                    },
                },
            )
        if resp.status_code == 200:
            data = resp.json()
            text = (data.get("result", {}).get("message", {})
                    .get("content", [{}])[0].get("text", ""))
            if text:
                return {"available": True, "reply": text.strip(), "error": None, "provider": "puter"}
    except Exception as exc:
        logger.error(f"[puter] chat error: {exc}")
    return None

async def chat(
    message: str,
    history: Optional[List[Dict]] = None,
    context: Optional[str] = None,
) -> Dict:
    """Main entry point — cascade through providers until one succeeds."""
    history = history or []
    if not message.strip():
        return {"available": True, "reply": "Please enter a question.", "error": None}

    # Try providers in order: Gemini → Claude → Grok → Puter (free)
    for try_fn in [_try_gemini, _try_claude, _try_grok, _try_puter]:
        result = await try_fn(message, history, context)
        if result and result.get("available"):
            logger.info(f"[chat] answered by {result.get('provider', 'unknown')}")
            return result

    # All providers failed
    return {
        "available": False,
        "reply": (
            "All AI providers are currently unavailable. "
            "Please try again in a moment, or add GEMINI_API_KEY in Replit Secrets."
        ),
        "error": "All providers failed",
    }
```

```python
async def fetch_injuries_for_match(self, home_team: str, away_team: str) -> List[Dict]:
    """
    Fetch injury data from TheSportsDB (free API key=3) for both teams.
    Falls back to empty list gracefully — never crashes the page.
    """
    injuries = []
    for team_name in [home_team, away_team]:
        try:
            async with httpx.AsyncClient(timeout=8) as client:
                # Search for team
                search_url = f"https://www.thesportsdb.com/api/v1/json/3/searchteams.php?t={team_name}"
                r = await client.get(search_url)
                if r.status_code != 200:
                    continue
                teams = r.json().get("teams") or []
                if not teams:
                    continue
                team_id = teams[0].get("idTeam")
                if not team_id:
                    continue

                # Get next event (to confirm team is active)
                next_url = f"https://www.thesportsdb.com/api/v1/json/3/eventsnext.php?id={team_id}"
                r2 = await client.get(next_url)
                # TheSportsDB free tier does not have injury data
                # Return a structured "no data" instead of empty
                injuries.append({
                    "team": team_name,
                    "player": "Injury data",
                    "status": "Not available for this league",
                    "detail": "Injury data requires a premium data provider. Configure SPORTRADAR_API_KEY or API_FOOTBALL_KEY for live injury reports.",
                    "source": "thesportsdb",
                })
        except Exception as exc:
            logger.warning(f"[injury] fetch failed for {team_name}: {exc}")
            continue
    return injuries
```
