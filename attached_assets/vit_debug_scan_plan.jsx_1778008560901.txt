const layerColors = SCAN_PLAN.layers.reduce((a, l) => ({ ...a, [l.id]: l.color }), {});

function CheckRow({ check, layerColor }) {
  const [open, setOpen] = useState(false);
  const [copied, setCopied] = useState(false);

  const copy = () => {
    navigator.clipboard.writeText(check.cmd);
    setCopied(true);
    setTimeout(() => setCopied(false), 1500);
  };

  return (
    <div className="border-b border-white/5 last:border-0">
      <button
        onClick={() => setOpen(!open)}
        className="w-full text-left px-4 py-3 flex items-start gap-3 hover:bg-white/5 transition"
      >
        <span className="text-xs font-mono mt-0.5 shrink-0" style={{ color: layerColor }}>{check.id}</span>
        <span className="flex-1 text-sm text-gray-300">{check.q}</span>
        <span className="text-gray-600 text-xs mt-0.5">{open ? "▲" : "▼"}</span>
      </button>
      {open && (
        <div className="px-4 pb-3 space-y-2 ml-10">
          <div className="text-xs text-gray-500 italic">{check.why}</div>
          <div className="relative">
            <pre className="bg-black/60 rounded-lg px-3 py-2 text-xs font-mono overflow-auto"
              style={{ color: layerColor, maxHeight: 120 }}>
              {check.cmd}
            </pre>
            <button
              onClick={copy}
              className="absolute top-1.5 right-2 text-xs px-2 py-0.5 rounded font-bold transition text-black"
              style={{ background: copied ? "#10b981" : layerColor }}
            >
              {copied ? "✓" : "copy"}
            </button>
          </div>
        </div>
      )}
    </div>
  );
}
