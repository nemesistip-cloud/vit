const severityColors = {
  CRITICAL: { bg: "bg-red-950", border: "border-red-500", badge: "bg-red-500", text: "text-red-400" },
  HIGH: { bg: "bg-orange-950", border: "border-orange-500", badge: "bg-orange-500", text: "text-orange-400" },
  MEDIUM: { bg: "bg-yellow-950", border: "border-yellow-600", badge: "bg-yellow-500", text: "text-yellow-400" },
};

const priorityColors = {
  P0: { badge: "bg-red-600", text: "text-red-300" },
  P1: { badge: "bg-orange-600", text: "text-orange-300" },
  P2: { badge: "bg-blue-700", text: "text-blue-300" },
};

function Badge({ label, color }) {
  return (
    <span className={`inline-flex items-center px-2 py-0.5 rounded text-xs font-bold text-white ${color}`}>
      {label}
    </span>
  );
}

function Card({ item, type }) {
  const [open, setOpen] = useState(false);
  const c = type === "security" ? severityColors[item.severity] : priorityColors[item.priority];
  return (
    <div
      className={`rounded-lg border ${c.border} ${c.bg} mb-3 overflow-hidden transition-all duration-200`}
      style={{ background: "rgba(0,0,0,0.4)" }}
    >
      <button
        className="w-full text-left px-4 py-3 flex items-center gap-3 hover:bg-white/5 transition"
        onClick={() => setOpen(!open)}
      >
        <Badge
          label={type === "security" ? item.severity : item.priority}
          color={c.badge}
        />
        <span className="font-mono text-xs text-gray-500">{item.id}</span>
        <span className="flex-1 font-semibold text-gray-100 text-sm">{item.title}</span>
        <span className={`text-lg ${open ? "rotate-180" : ""} transition-transform`}>▾</span>
      </button>
      {open && (
        <div className="px-4 pb-4 border-t border-white/10 pt-3 space-y-3">
          <div>
            <div className="text-xs text-gray-500 uppercase tracking-wider mb-1">Location</div>
            <code className="text-xs text-cyan-400 font-mono bg-black/40 px-2 py-1 rounded block">
              {item.location}
            </code>
          </div>
          <div>
            <div className="text-xs text-gray-500 uppercase tracking-wider mb-1">Issue</div>
            <p className="text-sm text-gray-300">{item.detail}</p>
          </div>
          <div>
            <div className="text-xs text-gray-500 uppercase tracking-wider mb-1">Fix</div>
            <p className="text-sm text-green-300">{item.fix}</p>
          </div>
        </div>
      )}
    </div>
  );
}

          </div>

        {/* Tab: Missing Features */}
        {tab === "missing" && (
          <div className="space-y-6">
            {data.missingFeatures.map(cat => (
              <div key={cat.category} className="rounded-lg border border-white/10 bg-black/30 overflow-hidden">
                <div className="px-4 py-3 bg-white/5 border-b border-white/10">
                  <span className="font-bold text-gray-200">{cat.category}</span>
                  <span className="ml-2 text-xs text-gray-500">{cat.items.length} items</span>
                </div>
                <div className="divide-y divide-white/5">
                  {cat.items.map(item => (
                    <div key={item.name} className="px-4 py-3 flex items-start gap-3">
                      <div className="flex-1">
                        <div className="text-sm font-semibold text-gray-200">{item.name}</div>
                        <div className="text-xs text-gray-500 mt-0.5">{item.impact}</div>
                      </div>
                      <span className={`text-xs px-2 py-0.5 rounded font-mono whitespace-nowrap ${
                        item.status.includes("missing") || item.status.includes("only") || item.status.includes("never")
                          ? "bg-red-900/60 text-red-300"
                          : item.status.includes("complete") || item.status.includes("exists")
                          ? "bg-yellow-900/60 text-yellow-300"
                          : "bg-gray-800 text-gray-400"
                      }`}>
                        {item.status}
                      </span>
                    </div>
                  ))}
                </div>
              </div>
            ))}
          </div>
        )}

        {/* Tab: Ecosystem */}
        {tab === "ecosystem" && (
          <div className="space-y-4">
            <div className="text-xs text-gray-500 mb-4">
              Prioritized implementation roadmap — each phase unlocks the next
            </div>
            {data.ecosystemUpgrades.map((phase, idx) => (
              <div key={phase.phase} className="rounded-lg border overflow-hidden" style={{ borderColor: phase.color + "55" }}>
                <div className="px-4 py-3 flex items-center gap-3" style={{ background: phase.color + "18" }}>
                  <span className="text-2xl font-black" style={{ color: phase.color }}>{idx + 1}</span>
                  <div>
                    <div className="font-bold text-gray-100">{phase.phase}</div>
                    <div className="text-xs text-gray-500">{phase.items.length} action items</div>
                  </div>
                </div>
                <div className="px-4 py-3 space-y-2">
                  {phase.items.map((item, i) => (
                    <div key={i} className="flex items-start gap-2">
                      <span style={{ color: phase.color }} className="mt-0.5 text-xs">▶</span>
                      <span className="text-sm text-gray-300">{item}</span>
                    </div>
                  ))}
                </div>
              </div>
            ))}

            {/* UX upgrade summary */}
            <div className="mt-6 rounded-lg border border-cyan-900 bg-cyan-950/30 px-4 py-4">
              <div className="font-bold text-cyan-300 mb-3">🎯 User Experience Priority Upgrades</div>
              <div className="grid grid-cols-1 md:grid-cols-2 gap-3 text-sm">
                {[
                  { title: "Real-Time Notifications", desc: "Connect frontend WebSocket → instant push on predictions, results, rewards" },
                  { title: "2FA Onboarding Flow", desc: "Step-by-step React modal with QR scanner + backup codes" },
                  { title: "Offerwall Reward Hub", desc: "User dashboard: earn VITCoin, track offers, see history" },
                  { title: "Mobile PWA", desc: "Add manifest.json + service worker for add-to-homescreen experience" },
                  { title: "Prediction Confidence UI", desc: "Visual probability bars, model consensus breakdown, Kelly stake guidance" },
                  { title: "Live Score Push", desc: "WebSocket live score updates on active matches → no refresh needed" },
                ].map(u => (
                  <div key={u.title} className="bg-black/30 rounded p-3 border border-cyan-900/50">
                    <div className="font-semibold text-cyan-200 text-xs uppercase tracking-wide">{u.title}</div>
                    <div className="text-gray-400 text-xs mt-1">{u.desc}</div>
                  </div>
                ))}
              </div>
            </div>
          </div>
        )}

        {/* Footer */}
        <div className="mt-8 border-t border-white/10 pt-4 text-xs text-gray-600 flex justify-between">
          <span>VIT OS Security & Engineering Audit · {data.meta.audited}</span>
          <span>{data.securityIssues.length + data.engineeringGaps.length} total findings</span>
        </div>
      </div>
    </div>
  );
}
