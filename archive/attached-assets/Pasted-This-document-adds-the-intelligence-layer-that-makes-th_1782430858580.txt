```python
class xGDataResolver:
    """
    Resolves xG data from the best available source per fixture.
    Never blocks certification if xG is unavailable — degrades gracefully.
    """

    async def resolve(self, fixture_id: int) -> xGDataPacket | None:

        # Tier 1: iSports API — check if xG included in match stats
        xg = await self._from_isports(fixture_id)
        if xg and xg.confidence >= 0.80:
            return xg.with_source("isports")

        # Tier 2: Derive from shot/possession data in existing fixture records
        # VIT already stores match stats — xG can be approximated
        xg = await self._derive_from_shots(fixture_id)
        if xg:
            return xg.with_source("derived").with_confidence(0.65)

        # Tier 3: Use league-average xG profiles as a prior
        # "In EPL, average home xG is 1.52, away 1.18 — apply as baseline"
        xg = await self._league_prior(fixture_id)
        if xg:
            return xg.with_source("prior").with_confidence(0.45)

        # No xG available — signal density score is penalised by 8 pts
        # but certification is NOT blocked
        return None


    async def _derive_from_shots(self, fixture_id: int) -> xGDataPacket | None:
        """
        Approximate xG from shot location + type data when direct feed unavailable.
        Uses a simplified Dixon-Coles-inspired model stored internally.

        Shot location zones (if available):
          6-yard box:     0.78 xG/shot
          Penalty spot:   0.76 xG/shot
          Inside box:     0.18 xG/shot
          Outside box:    0.04 xG/shot
          Header:         0.09 xG/shot (location-independent approximation)
        """
        shots = await db.get_fixture_shots(fixture_id)
        if not shots:
            return None

        home_xg = sum(XG_ZONE_WEIGHTS[s.zone] for s in shots if s.team == "home")
        away_xg = sum(XG_ZONE_WEIGHTS[s.zone] for s in shots if s.team == "away")

        return xGDataPacket(home_xg=home_xg, away_xg=away_xg)
```

### 2. Monte Carlo Simulation Engine

The document names this but doesn't implement it. Here's the full contract:

```python
class MonteCarloSimulator:
    """
    Runs N match simulations using Poisson goal-scoring distributions
    parameterised by xG values and attack/defence ratings.

    Output: probability distribution over outcomes,
            not just a point estimate.
    """

    DEFAULT_SIMULATIONS = 10_000

    def __init__(self, n_simulations: int = DEFAULT_SIMULATIONS):
        self.n = n_simulations

    def run(
        self,
        home_xg: float,
        away_xg: float,
        home_attack_rating: float,
        away_attack_rating: float,
        home_defence_rating: float,
        away_defence_rating: float,
    ) -> SimulationResult:
        import numpy as np
        from scipy.stats import poisson

        # Adjusted scoring rates — Dixon-Coles correction
        home_lambda = (
            home_xg
            * home_attack_rating
            * away_defence_rating    # strong away defence suppresses home scoring
        )
        away_lambda = (
            away_xg
            * away_attack_rating
            * home_defence_rating
        )

        # Vectorised simulation — no Python loop, memory-efficient
        home_goals = np.random.poisson(home_lambda, self.n)
        away_goals = np.random.poisson(away_lambda, self.n)

        home_wins = np.sum(home_goals > away_goals)
        draws     = np.sum(home_goals == away_goals)
        away_wins = np.sum(home_goals < away_goals)

        # BTTS and goals markets from simulation
        btts_yes  = np.sum((home_goals > 0) & (away_goals > 0))
        over_2_5  = np.sum((home_goals + away_goals) > 2)
        under_2_5 = np.sum((home_goals + away_goals) <= 2)

        total = self.n
        return SimulationResult(
            home_win_prob  = home_wins / total,
            draw_prob      = draws / total,
            away_win_prob  = away_wins / total,
            btts_yes_prob  = btts_yes / total,
            over_2_5_prob  = over_2_5 / total,
            under_2_5_prob = under_2_5 / total,
            home_lambda    = home_lambda,
            away_lambda    = away_lambda,
            simulations_run = self.n,
            # Distribution percentiles for UI display
            home_goals_p25 = float(np.percentile(home_goals, 25)),
            home_goals_p50 = float(np.percentile(home_goals, 50)),
            home_goals_p75 = float(np.percentile(home_goals, 75)),
        )
```

```python
    # Agreement bonus: simulation agrees with ensemble
    sim_agreement_bonus = 0.0
    if sim_prob_for_outcome >= 0.60:
        sim_agreement_bonus = 10.0    # Strong simulation backing
    elif sim_prob_for_outcome >= 0.50:
        sim_agreement_bonus = 5.0     # Moderate backing
    elif sim_prob_for_outcome < 0.40:
        sim_agreement_bonus = -15.0   # Simulation contradicts ensemble — penalise hard

    final_score = min(100, ensemble_result.score + sim_agreement_bonus)

    return ensemble_result.with_updates(
        score=final_score,
        simulation_result=sim_result,
        simulation_agreement=sim_prob_for_outcome,
    )
```

### 3. SHAP Explainability — Production Contract

The document says "TreeExplainer" without specifying which models it applies to. Here's the real implementation:

```python
class SHAPExplainabilityEngine:
    """
    Generates SHAP explanations for ensemble predictions.

    Model compatibility:
    - Tree-based models (XGBoost, LightGBM, RandomForest): TreeExplainer (fast, exact)
    - Neural models (LSTM, MLP): DeepExplainer (approximation)
    - Others: KernelExplainer (slow — only for low-frequency explanations)

    Render constraint: SHAP is computed asynchronously in the ai_core queue,
    never on the API request path.
    """

    def __init__(self, model_registry: dict):
        self.explainers = {}
        for model_id, model in model_registry.items():
            self.explainers[model_id] = self._build_explainer(model_id, model)

    def _build_explainer(self, model_id: str, model):
        import shap
        model_type = MODEL_TYPE_REGISTRY[model_id]

        if model_type in ("xgboost", "lightgbm", "random_forest"):
            return shap.TreeExplainer(model)
        elif model_type in ("lstm", "mlp", "transformer"):
            return shap.DeepExplainer(model, background_data=self._get_background())
        else:
            return shap.KernelExplainer(model.predict, self._get_background())

    def explain_fixture(
        self,
        fixture_features: dict,
        target_model_ids: list[str] | None = None
    ) -> SHAPExplanation:
        """
        Run SHAP on the models that voted for the consensus outcome.
        Only explain agreeing models — dissenting models are captured
        in conflict flags, not SHAP.
        """
        import shap
        import numpy as np

        model_ids = target_model_ids or list(self.explainers.keys())
        feature_array = self._to_feature_array(fixture_features)

        all_shap_values = []
        for model_id in model_ids:
            explainer = self.explainers[model_id]
            shap_vals = explainer.shap_values(feature_array)
            all_shap_values.append(shap_vals)

        # Aggregate across models — mean SHAP value per feature
        mean_shap = np.mean(all_shap_values, axis=0).flatten()
        feature_names = list(fixture_features.keys())

        # Top 5 positive contributors, top 3 negative
        ranked = sorted(
            zip(feature_names, mean_shap),
            key=lambda x: abs(x[1]),
            reverse=True
        )

        return SHAPExplanation(
            top_positive_drivers=[
                (name, float(val)) for name, val in ranked
                if val > 0
            ][:5],
            top_negative_drivers=[
                (name, float(val)) for name, val in ranked
                if val < 0
            ][:3],
            feature_importance_map=dict(zip(feature_names, mean_shap.tolist())),
            base_value=float(np.mean([e.expected_value for e in
                                      self.explainers.values()
                                      if hasattr(e, 'expected_value')])),
            consistency_score=self._compute_consistency(all_shap_values),
        )

    def _compute_consistency(self, all_shap_values) -> float:
        """
        Measures how consistently models agree on WHICH features drive the prediction.
        Low consistency = models agree on outcome but disagree on why.
        That's a certification red flag even if consensus score is high.
        """
        import numpy as np
        if len(all_shap_values) < 2:
            return 1.0

        # Rank correlation across model SHAP vectors
        from scipy.stats import spearmanr
        correlations = []
        flat = [v.flatten() for v in all_shap_values]
        for i in range(len(flat)):
            for j in range(i + 1, len(flat)):
                corr, _ = spearmanr(flat[i], flat[j])
                correlations.append(corr)

        return float(np.mean(correlations))
```

```python
class LIMEBinnedNarrativeEngine:
    """
    Generates human-readable prediction narratives using LIME
    with domain-aligned binning.

    Binning strategy: domain-aligned (not quantile/KMeans) because
    football has meaningful thresholds (e.g. win rate above 60% IS
    a high-form team — that's not a statistical artifact).
    """

    DOMAIN_BINS = {
        "home_win_rate":         [0.0, 0.40, 0.55, 0.70, 1.0],
        "away_win_rate":         [0.0, 0.30, 0.45, 0.60, 1.0],
        "btts_rate_home":        [0.0, 0.45, 0.60, 0.75, 1.0],
        "btts_rate_away":        [0.0, 0.45, 0.60, 0.75, 1.0],
        "avg_goals_scored_home": [0.0, 1.0,  1.5,  2.0,  5.0],
        "xg_home":               [0.0, 0.8,  1.2,  1.8,  4.0],
        "xg_away":               [0.0, 0.7,  1.1,  1.6,  4.0],
        "days_since_last_match": [0,   4,    7,    14,   90  ],
        "h2h_home_win_rate":     [0.0, 0.30, 0.50, 0.70, 1.0],
    }

    BIN_LABELS = {
        "home_win_rate":         ["Low", "Below Average", "Above Average", "High"],
        "xg_home":               ["Very Low", "Low", "Moderate", "High"],
        "days_since_last_match": ["Recent", "Normal Rest", "Well Rested", "Long Layoff"],
    }

    def explain_for_certificate(
        self,
        fixture_features: dict,
        consensus_outcome: str,
        shap_top_drivers: list[tuple]
    ) -> CertificateNarrative:
        """
        Generates the user-facing certificate narrative.
        Uses SHAP top drivers as anchor points, bins them,
        then generates readable sentences.
        """

        narrative_parts = []
        supporting_facts = []

        for feature_name, shap_value in shap_top_drivers[:3]:
            raw_value = fixture_features.get(feature_name)
            if raw_value is None:
                continue

            bin_label = self._get_bin_label(feature_name, raw_value)
            direction = "supports" if shap_value > 0 else "weakens"
            strength  = "strongly" if abs(shap_value) > 0.15 else "moderately"

            narrative_parts.append(
                f"{self._feature_display_name(feature_name)} "
                f"({bin_label}) {strength} {direction} this prediction"
            )

        # Compose final certificate narrative
        outcome_display = self._outcome_to_english(consensus_outcome)
        narrative = (
            f"VIT Intelligence certifies {outcome_display}. "
            + ". ".join(narrative_parts) + ". "
            + self._add_simulation_context(fixture_features)
        )

        return CertificateNarrative(
            summary=narrative,
            key_drivers=narrative_parts,
            outcome_display=outcome_display,
            confidence_statement=self._confidence_statement(shap_top_drivers),
        )

    def _confidence_statement(self, drivers: list) -> str:
        avg_abs_shap = sum(abs(v) for _, v in drivers) / max(len(drivers), 1)
        if avg_abs_shap > 0.20:
            return "High-conviction signal — multiple strong drivers aligned."
        elif avg_abs_shap > 0.10:
            return "Moderate conviction — primary driver identified."
        else:
            return "Signal present but diffuse — lower confidence tier."

    def _add_simulation_context(self, features: dict) -> str:
        sim = features.get("simulation_result")
        if not sim:
            return ""
        outcome = features.get("consensus_outcome")
        prob = {
            "home_win": sim.home_win_prob,
            "draw":     sim.draw_prob,
            "away_win": sim.away_win_prob,
            "btts_yes": sim.btts_yes_prob,
            "over_2.5": sim.over_2_5_prob,
        }.get(outcome, 0)
        return f"10,000 match simulations place probability at {prob:.0%}."
```

class RolloverChainAggregate:
    """
    The Event Sourcing aggregate root for a RolloverChain.

    Rules:
    - State is NEVER mutated directly
    - All changes go through apply(event)
    - apply() is called by both command handlers (new events)
      and the event store loader (replaying history)
    - Commands validate business rules BEFORE emitting events
    """

    def __init__(self, chain_id: int):
        self.chain_id          = chain_id
        self.status            = "UNINITIALISED"
        self.legs              = {}
        self.insurance_count   = 0
        self.insurance_used    = 0
        self.legs_won          = 0
        self.version           = 0      # Optimistic concurrency version
        self._pending_events   = []     # Events not yet persisted

    # ── Command handlers ──────────────────────────────────────────────────

    def create(self, user_id, legs, config):
        self._assert_state("UNINITIALISED")
        self._emit(ChainCreatedEvent(
            chain_id=self.chain_id,
            user_id=user_id,
            legs=[l.to_dict() for l in legs],
            config=config.to_dict(),
        ))

            raise LegAlreadySettled(leg_id)
